TMap<FIntVector, FGameplayTag> UVoxelSelectionTool::CalculateSelectionWithTypes(
    const FVoxelEditRequest& Request,
    float BlockSize,
    UVoxelBlockWorldSubsystem* BlockWorld)
{
    TMap<FIntVector, FGameplayTag> Result;

    if (!BlockWorld)
    {
        return Result;
    }

    // 1. 计算形状选区
    TArray<FIntVector> ShapePositions = CalculateSelection(Request, BlockSize);

    // 2. 查询每个位置的体素类�?
    FGameplayTag AirTag = FGameplayTag::RequestGameplayTag(FName("Voxel.Block.Air"));

    for (const FIntVector& Position : ShapePositions)
    {
        FGameplayTag BlockType = BlockWorld->GetVoxelType(Position);

        // 过滤空气方块
        if (!BlockType.IsValid() || BlockType == AirTag)
        {
            continue;
        }

        Result.Add(Position, BlockType);
    }

    return Result;
}

TMap<FIntVector, FGameplayTag> UVoxelSelectionTool::CalculateSelectionWithTypes(
    const FVoxelEditRequest& Request,
    float BlockSize,
    const TMap<FIntVector, FGameplayTag>& ExistingSelection,
    ESelectionOperationType OperationType,
    UVoxelBlockWorldSubsystem* BlockWorld)
{
    // 1. 计算新的形状选区（带类型�?
    TMap<FIntVector, FGameplayTag> NewSelection = CalculateSelectionWithTypes(
        Request, BlockSize, BlockWorld);

    // 2. 根据操作类型应用布尔运算
    TMap<FIntVector, FGameplayTag> Result;

    switch (OperationType)
    {
    case ESelectionOperationType::New:
        Result = NewSelection;
        break;

    case ESelectionOperationType::Add:
        // 并集：合并两个选区
        Result.Append(ExistingSelection);
        Result.Append(NewSelection);
        break;

    case ESelectionOperationType::Remove:
        // 差集：从现有选区移除新选区
        Result = ExistingSelection;
        for (const auto& Pair : NewSelection)
        {
            Result.Remove(Pair.Key);
        }
        break;

    case ESelectionOperationType::Intersect:
        // 交集：只保留两个选区共有的部�?
        for (const auto& Pair : ExistingSelection)
        {
            if (NewSelection.Contains(Pair.Key))
            {
                Result.Add(Pair.Key, Pair.Value);
            }
        }
        break;

    case ESelectionOperationType::Replace:
        Result = NewSelection;
        break;

    case ESelectionOperationType::Invert:
        // 反选：需要在包围盒范围内反转
        // TODO: 实现反选逻辑
        Result = NewSelection;
        break;
    }

    return Result;
}

bool UVoxelControllerInteractComponent::CreateTemplateFromSelection(
    const FVoxelSelection& Selection,
    FString TemplateName,
    FString Description,
    TArray<FGameplayTag> Tags,
    UVoxelBuildingTemplateAsset*& OutTemplate)
{
    if (Selection.IsEmpty())
    {
        OutTemplate = nullptr;
        return false;
    }

    // 步骤1：创建新模板资产
    OutTemplate = NewObject<UVoxelBuildingTemplateAsset>(this);

    // 步骤2：计算包围盒
    FIntVector MinCorner = FIntVector(TNumericLimits<int32>::Max());
    FIntVector MaxCorner = FIntVector(TNumericLimits<int32>::Min());

    for (const auto& Pair : Selection.SelectedVoxels)
    {
        MinCorner = FIntVector(
            FMath::Min(MinCorner.X, Pair.Key.X),
            FMath::Min(MinCorner.Y, Pair.Key.Y),
            FMath::Min(MinCorner.Z, Pair.Key.Z)
        );
        MaxCorner = FIntVector(
            FMath::Max(MaxCorner.X, Pair.Key.X),
            FMath::Max(MaxCorner.Y, Pair.Key.Y),
            FMath::Max(MaxCorner.Z, Pair.Key.Z)
        );
    }

    OutTemplate->Size = MaxCorner - MinCorner + FIntVector(1, 1, 1);

    // 步骤3：计算锚点偏移（底面中心�?
    OutTemplate->AnchorOffset = FIntVector(
        (OutTemplate->Size.X - 1) / 2,
        (OutTemplate->Size.Y - 1) / 2,
        0
    );

    // 步骤4：构建调色板（去�?+ 排序�?
    TSet<FGameplayTag> UniqueTags;
    FGameplayTag AirTag = FGameplayTag::RequestGameplayTag(FName("Voxel.Block.Air"));

    for (const auto& Pair : Selection.SelectedVoxels)
    {
        FGameplayTag Tag = Pair.Value;

        // 过滤空气方块
        if (Tag == AirTag)
        {
            continue;
        }

        // 过滤脚手架层�?
        if (Tag.RequestTagExplicitlyMatches(FGameplayTag::RequestGameplayTag(FName("Voxel.Block.Scaffolding"))))
        {
            continue;
        }

        UniqueTags.Add(Tag);
    }

    // 按TagName排序（确保跨平台一致性）
    TArray<FGameplayTag> SortedTags = UniqueTags.Array();
    SortedTags.Sort([](const FGameplayTag& A, const FGameplayTag& B)
    {
        return A.GetTagName().CompareTo(B.GetTagName()) < 0;
    });

    OutTemplate->Palette = SortedTags;

    // 构建Tag到索引的映射
    TMap<FGameplayTag, int32> TagToIndex;
    for (int32 i = 0; i < OutTemplate->Palette.Num(); i++)
    {
        TagToIndex.Add(OutTemplate->Palette[i], i);
    }

    // 步骤5：转换体素数据（底面中心为原点）
    FIntVector Origin = MinCorner + OutTemplate->AnchorOffset;

    for (const auto& Pair : Selection.SelectedVoxels)
    {
        FIntVector WorldPos = Pair.Key;
        FGameplayTag BlockType = Pair.Value;

        // 计算相对坐标
        FIntVector RelativePos = WorldPos - Origin;

        // 查找调色板索�?
        int32* PaletteIndexPtr = TagToIndex.Find(BlockType);
        if (PaletteIndexPtr)
        {
            OutTemplate->VoxelData.Add(RelativePos, *PaletteIndexPtr);
        }
    }

    // 步骤6：填充元数据
    OutTemplate->TemplateName = TemplateName;
    OutTemplate->Description = Description;
    OutTemplate->Tags = Tags;
    OutTemplate->AuthorName = TEXT("Unknown"); // TODO: 从玩家数据获�?
    OutTemplate->CreationDate = FDateTime::Now();
    OutTemplate->BlockCount = OutTemplate->VoxelData.Num();

    return true;
}
