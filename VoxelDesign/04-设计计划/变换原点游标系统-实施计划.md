# 变换原点游标系统 实施计划

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 引入双模式变换原点（物体中心 / 游标）+ 双模式坐标空间（绝对坐标系 / 局部轴），实现游标模式下轨道旋转，M4b 面板控制 + 3D 游标 Actor 独立交互。

**Architecture:** 所有新增数据（PivotMode、TransformSpace、PivotWorld）存储在 `FVoxelGeometryDesc` 内，不新增结构体。`Center` 缓存字段删除改为实时计算 `GetGeometricCenter()`。新增 `GetPivotWorld()` 作为变换原点的唯一入口。游标 Actor 由 VisualUIComponent 管理，与 Gizmo 同级。

**Tech Stack:** Unreal Engine 5.7 C++, Blueprint (WBP_M4b_GeometryParamsModule)

**前置依赖:** [交互系统-射线检测与状态驱动重构](./交互系统-射线检测与状态驱动重构-实施计划.md) 已完成。

```cpp
    FVector V1, V2;
    EVoxelTransformSpace Space = Geo.TransformSpace;

    if (Space == EVoxelTransformSpace::Local)
    {
        // 局部轴模式：转到物体局部空间投影
        FQuat InvRot = Geo.Rotation.Quaternion().Inverse();
        FVector LocalFirst  = InvRot * PivotToFirst;
        FVector LocalSecond = InvRot * PivotToSecond;

        switch (AxisIndex)
        {
        case 0: // 局部 X 轴：投影到 YZ 平面
            V1 = FVector(0, LocalFirst.Y, LocalFirst.Z);
            V2 = FVector(0, LocalSecond.Y, LocalSecond.Z);
            break;
        case 1: // 局部 Y 轴：投影到 XZ 平面
            V1 = FVector(LocalFirst.X, 0, LocalFirst.Z);
            V2 = FVector(LocalSecond.X, 0, LocalSecond.Z);
            break;
        case 2: // 局部 Z 轴：投影到 XY 平面
            V1 = FVector(LocalFirst.X, LocalFirst.Y, 0);
            V2 = FVector(LocalSecond.X, LocalSecond.Y, 0);
            break;
        }
    }
    else
    {
        // 绝对坐标系：投影到世界 X/Y/Z 平面（当前逻辑）
        switch (AxisIndex)
        {
        case 0:
            V1 = FVector(0, PivotToFirst.Y, PivotToFirst.Z);
            V2 = FVector(0, PivotToSecond.Y, PivotToSecond.Z);
            break;
        case 1:
            V1 = FVector(PivotToFirst.X, 0, PivotToFirst.Z);
            V2 = FVector(PivotToSecond.X, 0, PivotToSecond.Z);
            break;
        case 2:
            V1 = FVector(PivotToFirst.X, PivotToFirst.Y, 0);
            V2 = FVector(PivotToSecond.X, PivotToSecond.Y, 0);
            break;
        }
    }
```

**Step 3: 增加轨道旋转**

替换 L205-229 的旋转应用逻辑：

```cpp
    // 计算角度增量
    const float Angle1 = FMath::Atan2(V1.Y, V1.X);
    const float Angle2 = FMath::Atan2(V2.Y, V2.X);
    const float DeltaAngle = FMath::RadiansToDegrees(Angle2 - Angle1);

    // === 步骤A: 构造增量旋转 ===
    FRotator DeltaRotation = FRotator::ZeroRotator;
    switch (AxisIndex)
    {
    case 0: DeltaRotation.Pitch = DeltaAngle; break;
    case 1: DeltaRotation.Yaw   = DeltaAngle; break;
    case 2: DeltaRotation.Roll  = DeltaAngle; break;
    }

    // === 步骤B: 轨道旋转（Cursor 模式下更新 Box 位置）===
    if (Geo.PivotMode == EVoxelPivotMode::Cursor)
    {
        FVector GeoCenter = Geo.GetGeometricCenter(BlockLength);
        FVector Offset = GeoCenter - Pivot;
        FVector NewOffset = DeltaRotation.RotateVector(Offset);
        FVector NewGeoCenter = Pivot + NewOffset;
        FVector DeltaWorld = NewGeoCenter - GeoCenter;

        FIntVector DeltaIndex = UChunkUtilsFunction::WorldDeltaToIndexDelta(DeltaWorld, BlockLength);
        FVoxelGeometryDesc MutableGeo = Geo;
        MutableGeo.Box.MinIndex += DeltaIndex;
        MutableGeo.Box.MaxIndex += DeltaIndex;
        MutableGeo.Rotation += DeltaRotation;
        SC->SetTmpGeometry(MutableGeo);
    }
    else
    {
        // ObjectCenter 模式：原地旋转
        FRotator NewRot = Geo.Rotation + DeltaRotation;
        SC->SetTmpGeometryRotation(NewRot);
    }
```

```cpp
void UVoxelControllerVisualUIComponent::EnsureCursorActor()
{
    if (CursorActor && IsValid(CursorActor)) return;

    FActorSpawnParameters SpawnParams;
    SpawnParams.Owner = GetOwner();
    SpawnParams.SpawnCollisionHandlingOverride = ESpawnActorCollisionHandlingMethod::AlwaysSpawn;

    CursorActor = GetWorld()->SpawnActor<AVoxelPivotCursorActor>(
        AVoxelPivotCursorActor::StaticClass(), SpawnParams);
}

void UVoxelControllerVisualUIComponent::UpdateCursorActor(const FVector& PivotWorld, bool bVisible)
{
    if (!CursorActor || !IsValid(CursorActor)) return;

    CursorActor->SetCursorWorldLocation(PivotWorld);
    CursorActor->SetActorHiddenInGame(!bVisible);
}
```

```cpp
void UM4b_GeometryParamsModule::OnPivotModeChanged(FString SelectedItem, ESelectInfo::Type SelectionType)
{
    if (!StateComponent) return;

    EVoxelPivotMode NewMode = (SelectedItem == TEXT("游标"))
        ? EVoxelPivotMode::Cursor
        : EVoxelPivotMode::ObjectCenter;

    FVoxelGeometryDesc Geo = StateComponent->GetTmpGeometry();
    Geo.PivotMode = NewMode;
    StateComponent->SetTmpGeometry(Geo);
}

void UM4b_GeometryParamsModule::OnTransformSpaceChanged(FString SelectedItem, ESelectInfo::Type SelectionType)
{
    if (!StateComponent) return;

    EVoxelTransformSpace NewSpace = (SelectedItem == TEXT("局部轴"))
        ? EVoxelTransformSpace::Local
        : EVoxelTransformSpace::Global;

    FVoxelGeometryDesc Geo = StateComponent->GetTmpGeometry();
    Geo.TransformSpace = NewSpace;
    StateComponent->SetTmpGeometry(Geo);
}

void UM4b_GeometryParamsModule::OnCursorToCenterClicked()
{
    if (!StateComponent) return;

    FVoxelGeometryDesc Geo = StateComponent->GetTmpGeometry();
    // 获取 BlockLength
    float BlockLength = 100.0f;
    if (UVoxelGameInstance* GI = GetWorld()->GetGameInstance<UVoxelGameInstance>())
    {
        BlockLength = GI->WorldConfig.BlockLength;
    }
    Geo.PivotWorld = Geo.GetGeometricCenter(BlockLength);
    StateComponent->SetTmpGeometry(Geo);
}

void UM4b_GeometryParamsModule::OnCenterToCursorClicked()
{
    if (!StateComponent) return;

    FVoxelGeometryDesc Geo = StateComponent->GetTmpGeometry();
    float BlockLength = 100.0f;
    if (UVoxelGameInstance* GI = GetWorld()->GetGameInstance<UVoxelGameInstance>())
    {
        BlockLength = GI->WorldConfig.BlockLength;
    }

    // 移动物体使几何中心 = 游标位置
    FVector CurrentCenter = Geo.GetGeometricCenter(BlockLength);
    FVector Delta = Geo.PivotWorld - CurrentCenter;
    FIntVector DeltaIndex = UChunkUtilsFunction::WorldDeltaToIndexDelta(Delta, BlockLength);
    Geo.Box.MinIndex += DeltaIndex;
    Geo.Box.MaxIndex += DeltaIndex;

    StateComponent->SetTmpGeometry(Geo);
}
```
