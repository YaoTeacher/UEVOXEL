void UBuildPanelWidget::RegisterModule(UToolDetailModuleBase* Module)
{
    if (!Module || !ToolDetailPanel)
    {
        UE_LOG(LogTemp, Error, TEXT("BuildPanel: Failed to register module - null module or panel"));
        return;
    }

    // 设置 StateComponent 引用
    if (AVoxelPlayerController* PC = GetOwningPlayer<AVoxelPlayerController>())
    {
        Module->StateComponent = PC->StateComp;
    }

    // 添加到容�?
    ToolDetailPanel->AddChild(Module);
    RegisteredModules.Add(Module);

    UE_LOG(LogTemp, VeryVerbose, TEXT("BuildPanel: Module registered: %s"), *Module->GetName());
}

void UBuildPanelWidget::UpdateModuleVisibility(FGameplayTag NewToolTag)
{
    if (!NewToolTag.IsValid())
    {
        // 隐藏所有模�?
        for (UToolDetailModuleBase* Module : RegisteredModules)
        {
            if (Module)
            {
                Module->SetVisibility(ESlateVisibility::Collapsed);
            }
        }
        return;
    }

    // 遍历所有已注册模块
    for (UToolDetailModuleBase* Module : RegisteredModules)
    {
        if (!Module) continue;

        // 判断是否应该显示
        bool bShouldShow = Module->ShouldShowForTool(NewToolTag);

        if (bShouldShow)
        {
            Module->SetVisibility(ESlateVisibility::Visible);
            // �?StateComponent 更新模块数据
            if (AVoxelPlayerController* PC = GetOwningPlayer<AVoxelPlayerController>())
            {
                Module->UpdateFromToolState(PC->StateComp);
            }
        }
        else
        {
            Module->SetVisibility(ESlateVisibility::Collapsed);
        }
    }

    UE_LOG(LogTemp, VeryVerbose, TEXT("BuildPanel: Module visibility updated for tool: %s"),
        *NewToolTag.GetTagName().ToString());
}
